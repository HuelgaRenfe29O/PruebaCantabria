function mostrarHorario(){

    const sentido = document.getElementById("sentido").value;
    const visor = document.getElementById("visor");

        visor.innerHTML = `
            <img src="imagenes/C1_Santander_Reinosa_Pag1.jpg"
                 alt="C1 Santander Reinosa">
        `;
    }

    else if(sentido === "rs"){
        visor.innerHTML = `
            <img src="imagenes/C1_Reinosa_Santander_Pag2.jpg"
                 alt="C1 Reinosa Santander">
        `;
    }

    else{
        visor.innerHTML = "";
